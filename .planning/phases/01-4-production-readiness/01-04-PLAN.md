    // In-flight request tracking for serverless timeout awareness
    // Uses per-request finalized flag on response object to guarantee exactly-once decrement
    // Configurable via environment: DRAIN_TIMEOUT_MS (default 8000 for 10s Vercel timeout)
    
    let inFlightCount = 0;
    let isDraining = false;
    let drainStartTime = null;
    const DRAIN_TIMEOUT_MS = parseInt(process.env.DRAIN_TIMEOUT_MS || '8000', 10);
    
    function decrementInFlight(res) {
      // Per-request finalized guard: only decrement if this specific response hasn't finalized yet
      if (res.__drainFinalized) return;
      res.__drainFinalized = true;
      if (inFlightCount > 0) {
        inFlightCount--;
      }
    }
    
    export function requestDrainMiddleware(req, res, next) {
      // Skip health checks from drain tracking
      if (req.path === '/health' || req.path === '/api/health') {
        return next();
      }
    
      if (isDraining) {
        // Server is draining, reject new requests with 503
        const remainingMs = Math.max(0, DRAIN_TIMEOUT_MS - (Date.now() - drainStartTime));
        res.set('Retry-After', Math.ceil(remainingMs / 1000).toString());
        return res.status(503).json({
          error: 'Service Unavailable',
          message: 'Server is draining, please retry',
          retryAfter: Math.ceil(remainingMs / 1000)
        });
      }
    
      // Increment in-flight counter
      inFlightCount++;
    
      // Exactly-once decrement on response finish (successful completion)
      res.on('finish', () => decrementInFlight(res));
    
      // Exactly-once decrement on response close (client aborted, error, etc.)
      // The per-request guard ensures finish + close cannot double-decrement
      res.on('close', () => decrementInFlight(res));
    
      next();
    }
    
    export function startDrain() {
      if (isDraining) return; // Idempotent
      isDraining = true;
      drainStartTime = Date.now();
      console.log(`[DRAIN] Started draining at ${new Date().toISOString()}, in-flight: ${inFlightCount}`);
    }
    
    export function getDrainState() {
      return {
        isDraining,
        inFlightCount,
        drainStartTime,
        elapsedMs: drainStartTime ? Date.now() - drainStartTime : 0,
        timeoutMs: DRAIN_TIMEOUT_MS,
        remainingMs: drainStartTime ? Math.max(0, DRAIN_TIMEOUT_MS - (Date.now() - drainStartTime)) : DRAIN_TIMEOUT_MS
      };
    }
    
    export async function waitForDrain() {
      if (!isDraining) return { drained: true, inFlightCount: 0 };
    
      const startWait = Date.now();
      const checkInterval = 250;
    
      while (inFlightCount > 0) {
        const elapsed = Date.now() - startWait;
        if (elapsed >= DRAIN_TIMEOUT_MS) {
          console.log(`[DRAIN] Timeout reached (${DRAIN_TIMEOUT_MS}ms), ${inFlightCount} requests still in-flight`);
          return { drained: false, inFlightCount, timedOut: true };
        }
        await new Promise(resolve => setTimeout(resolve, checkInterval));
      }
    
      console.log(`[DRAIN] All requests drained in ${Date.now() - startWait}ms`);
      return { drained: true, inFlightCount: 0 };
    }

      return isDraining;
    }

      inFlightCount = 0;
      isDraining = false;
      drainStartTime = null;

    import { randomBytes } from 'crypto';
    import { env } from '../config/env.js';
    
    const CSRF_COOKIE_NAME = 'csrf_token';
    const CSRF_HEADER_NAME = 'x-csrf-token';
    const CSRF_COOKIE_MAX_AGE = 24 * 60 * 60 * 1000; // 24 hours
    
    function generateToken() {
      return randomBytes(32).toString('hex');
    }
    
    function getTokenFromRequest(req) {
      // Check header first (for API clients)
      const headerToken = req.headers[CSRF_HEADER_NAME] || req.headers['x-csrf-token'];
      if (headerToken) return headerToken;
      
      // Check cookie (for browser forms)
      const cookieToken = req.cookies?.[CSRF_COOKIE_NAME];
      if (cookieToken) return cookieToken;
      
      return null;
    }
    
    function validateOrigin(req) {
      const origin = req.headers.origin;
      if (!origin) return true; // Same-origin requests may not send Origin
      
      return env.frontendOrigins.some(allowed => {
        try {
          return new URL(origin).origin === new URL(allowed).origin;
        } catch {
          return false;
        }
      });
    }
    
    export function csrfMiddleware(req, res, next) {
      // Only apply to state-changing methods on /api routes
      if (!req.path.startsWith('/api')) return next();
      if (['GET', 'HEAD', 'OPTIONS'].includes(req.method)) return next();
      
      // Skip if origin not allowed (will be caught by CORS anyway)
      if (!validateOrigin(req)) {
        return res.status(403).json({
          success: false,
          error: { code: 'CSRF_ORIGIN_INVALID', message: 'Invalid origin' }
        });
      }
      
      // For browser requests, ensure CSRF cookie exists
      const isBrowserRequest = req.headers['user-agent']?.includes('Mozilla');
      const cookieToken = req.cookies?.[CSRF_COOKIE_NAME];
      
      if (isBrowserRequest && !cookieToken) {
        // Set CSRF cookie for browser
        const token = generateToken();
        res.cookie(CSRF_COOKIE_NAME, token, {
          httpOnly: false, // Must be readable by JS for double-submit
          secure: env.isProduction,
          sameSite: 'lax',
          maxAge: CSRF_COOKIE_MAX_AGE,
          path: '/'
        });
      }
      
      // Validate token for state-changing requests
      const requestToken = getTokenFromRequest(req);
      const expectedToken = cookieToken || req.cookies?.[CSRF_COOKIE_NAME];
      
      if (!requestToken || !expectedToken || requestToken !== expectedToken) {
        return res.status(403).json({
          success: false,
          error: { code: 'CSRF_TOKEN_INVALID', message: 'Invalid or missing CSRF token' }
        });
      }
      
      next();
    }
    
    export function setCsrfCookie(req, res) {
      const token = generateToken();
      res.cookie(CSRF_COOKIE_NAME, token, {
        httpOnly: false,
        secure: env.isProduction,
        sameSite: 'lax',
        maxAge: CSRF_COOKIE_MAX_AGE,
        path: '/'
      });
      return token;
    }
